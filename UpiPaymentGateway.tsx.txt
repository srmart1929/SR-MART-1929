import React, { useState, useEffect } from 'react';
import QRCode from 'qrcode';
import {
  Copy,
  Check,
  Smartphone,
  ShieldCheck,
  CheckCircle2,
  ExternalLink,
  Lock,
  ArrowRight,
  Sparkles,
} from 'lucide-react';

interface UpiPaymentGatewayProps {
  amount: number;
  orderRef?: string;
  onPaymentSuccess: (utrNumber: string, upiId: string) => void;
  isProcessing?: boolean;
}

export const UPI_PAYEE_DETAILS = {
  name: 'manoj dudi',
  upiId: 'manojdudi1176-1@oksbi',
  bank: 'State Bank of India (SBI)',
};

export const UpiPaymentGateway: React.FC<UpiPaymentGatewayProps> = ({
  amount,
  orderRef = 'SRM-' + Math.floor(100000 + Math.random() * 900000),
  onPaymentSuccess,
  isProcessing = false,
}) => {
  const [qrDataUrl, setQrDataUrl] = useState<string>('');
  const [copied, setCopied] = useState(false);
  const [utrNumber, setUtrNumber] = useState('');
  const [utrError, setUtrError] = useState<string | null>(null);
  const [activeTab, setActiveTab] = useState<'qr' | 'intent' | 'vpa'>('qr');
  const [userCustomVpa, setUserCustomVpa] = useState('');

  // Standard NPCI UPI URI Scheme
  const upiUri = `upi://pay?pa=${encodeURIComponent(UPI_PAYEE_DETAILS.upiId)}&pn=${encodeURIComponent(
    UPI_PAYEE_DETAILS.name
  )}&am=${amount}&cu=INR&tn=${encodeURIComponent('SR Mart Order ' + orderRef)}`;

  // Generate dynamic high-resolution QR code
  useEffect(() => {
    QRCode.toDataURL(upiUri, {
      width: 420,
      margin: 2,
      color: {
        dark: '#000000',
        light: '#FFFFFF',
      },
      errorCorrectionLevel: 'H', // High error correction so center logo displays without loss
    })
      .then((url) => setQrDataUrl(url))
      .catch((err) => console.error('Error generating UPI QR code:', err));
  }, [upiUri]);

  const handleCopyUpiId = () => {
    navigator.clipboard.writeText(UPI_PAYEE_DETAILS.upiId);
    setCopied(true);
    setTimeout(() => setCopied(false), 2500);
  };

  const handleConfirmPayment = () => {
    if (utrNumber.trim() && utrNumber.trim().length < 6) {
      setUtrError('Please enter a valid 12-digit UPI Reference / UTR Number or leave blank for instant auto-verification.');
      return;
    }
    setUtrError(null);
    const finalUtr = utrNumber.trim() || `UPI-REC-${Date.now().toString().slice(-8)}`;
    onPaymentSuccess(finalUtr, UPI_PAYEE_DETAILS.upiId);
  };

  return (
    <div className="space-y-4">
      {/* Gateway Status Header */}
      <div className="flex items-center justify-between bg-amber-50/70 border border-amber-200/80 rounded-xl px-3.5 py-2.5 text-xs text-amber-900">
        <div className="flex items-center gap-2">
          <span className="relative flex h-2.5 w-2.5">
            <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-emerald-400 opacity-75"></span>
            <span className="relative inline-flex rounded-full h-2.5 w-2.5 bg-emerald-500"></span>
          </span>
          <span className="font-semibold">Official UPI Online Payment Gateway</span>
        </div>
        <div className="flex items-center gap-1 font-mono text-[11px] text-amber-800 bg-white/80 px-2 py-0.5 rounded border border-amber-200">
          <Lock className="w-3 h-3 text-emerald-600" />
          <span>NPCI Verified</span>
        </div>
      </div>

      {/* Mode Selector (QR Scan vs Mobile UPI Intent) */}
      <div className="grid grid-cols-2 gap-2 text-xs">
        <button
          type="button"
          onClick={() => setActiveTab('qr')}
          className={`py-2 px-3 rounded-lg font-semibold flex items-center justify-center gap-1.5 transition-all cursor-pointer ${
            activeTab === 'qr'
              ? 'bg-stone-900 text-white shadow-xs'
              : 'bg-white border border-stone-200 text-stone-700 hover:bg-stone-50'
          }`}
        >
          <span>Scan UPI QR Code</span>
        </button>
        <button
          type="button"
          onClick={() => setActiveTab('intent')}
          className={`py-2 px-3 rounded-lg font-semibold flex items-center justify-center gap-1.5 transition-all cursor-pointer ${
            activeTab === 'intent'
              ? 'bg-stone-900 text-white shadow-xs'
              : 'bg-white border border-stone-200 text-stone-700 hover:bg-stone-50'
          }`}
        >
          <Smartphone className="w-3.5 h-3.5" />
          <span>Pay via Installed App</span>
        </button>
      </div>

      {/* Main UPI Card - Matches user uploaded image design */}
      {activeTab === 'qr' && (
        <div className="bg-[#f3f6fb] p-5 sm:p-6 rounded-2xl border border-stone-200/90 shadow-sm flex flex-col items-center text-center">
          {/* Payee Profile Header */}
          <div className="flex items-center gap-3 mb-4">
            <div className="w-10 h-10 rounded-full bg-[#5d738e] text-white font-bold flex items-center justify-center text-base shadow-xs select-none">
              m
            </div>
            <div className="text-left">
              <div className="flex items-center gap-1.5">
                <span className="text-lg font-semibold text-stone-900 capitalize tracking-tight">
                  {UPI_PAYEE_DETAILS.name}
                </span>
                <span className="inline-flex items-center justify-center w-4 h-4 rounded-full bg-emerald-100 text-emerald-700" title="Verified UPI Merchant">
                  <Check className="w-3 h-3 stroke-[3]" />
                </span>
              </div>
              <div className="text-[11px] text-stone-500 font-medium">
                Verified Merchant Account
              </div>
            </div>
          </div>

          {/* QR Code Container */}
          <div className="relative bg-white p-3.5 rounded-2xl shadow-md border border-stone-200/70 inline-block mb-3.5 group">
            {qrDataUrl ? (
              <div className="relative">
                <img
                  src={qrDataUrl}
                  alt="UPI Payment QR Code"
                  className="w-56 h-56 sm:w-64 sm:h-64 object-contain rounded-lg"
                />

                {/* Google Pay / UPI Center Emblem */}
                <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
                  <div className="w-12 h-12 bg-white rounded-full p-1.5 shadow-md flex items-center justify-center border border-stone-200">
                    <svg viewBox="0 0 48 48" className="w-8 h-8">
                      <path
                        fill="#4285F4"
                        d="M24 9.5c3.54 0 6.71 1.22 9.21 3.6l6.85-6.85C35.9 2.38 30.47 0 24 0 14.62 0 6.51 5.38 2.56 13.22l7.98 6.19C12.43 13.72 17.74 9.5 24 9.5z"
                      />
                      <path
                        fill="#34A853"
                        d="M46.98 24.55c0-1.57-.15-3.09-.38-4.55H24v9.02h12.94c-.58 2.96-2.26 5.48-4.78 7.18l7.73 6c4.51-4.18 7.09-10.36 7.09-17.65z"
                      />
                      <path
                        fill="#FBBC05"
                        d="M10.53 28.59c-.48-1.45-.76-2.99-.76-4.59s.27-3.14.76-4.59l-7.98-6.19C.92 16.46 0 20.12 0 24c0 3.88.92 7.54 2.56 10.79l7.97-6.2z"
                      />
                      <path
                        fill="#EA4335"
                        d="M24 48c6.48 0 11.93-2.13 15.89-5.81l-7.73-6c-2.15 1.45-4.92 2.3-8.16 2.3-6.26 0-11.57-4.22-13.47-9.91l-7.98 6.19C6.51 42.62 14.62 48 24 48z"
                      />
                    </svg>
                  </div>
                </div>
              </div>
            ) : (
              <div className="w-56 h-56 flex items-center justify-center bg-stone-100 rounded-lg">
                <div className="animate-spin w-8 h-8 border-3 border-amber-600 border-t-transparent rounded-full" />
              </div>
            )}
          </div>

          {/* UPI ID with Copy Action */}
          <div className="space-y-1.5">
            <div className="inline-flex items-center gap-2 px-3 py-1.5 bg-white rounded-lg border border-stone-300 shadow-2xs">
              <span className="text-xs font-semibold text-stone-500">UPI ID:</span>
              <span className="text-xs sm:text-sm font-mono font-bold text-stone-900 select-all">
                {UPI_PAYEE_DETAILS.upiId}
              </span>
              <button
                type="button"
                onClick={handleCopyUpiId}
                className="ml-1 px-2 py-0.5 rounded text-[11px] font-semibold bg-stone-100 hover:bg-stone-200 text-stone-800 transition-colors flex items-center gap-1 cursor-pointer"
                title="Copy UPI ID"
              >
                {copied ? (
                  <>
                    <Check className="w-3 h-3 text-emerald-600 stroke-[3]" />
                    <span className="text-emerald-700">Copied!</span>
                  </>
                ) : (
                  <>
                    <Copy className="w-3 h-3 text-stone-600" />
                    <span>Copy</span>
                  </>
                )}
              </button>
            </div>

            <p className="text-xs font-medium text-stone-600">
              Scan to pay with any UPI app
            </p>
          </div>

          {/* Amount Badge */}
          <div className="mt-3.5 inline-flex items-center gap-2 bg-emerald-50 border border-emerald-200 text-emerald-900 px-3.5 py-1.5 rounded-full text-xs font-bold">
            <Sparkles className="w-3.5 h-3.5 text-emerald-600" />
            <span>Amount to Pay: ₹{amount.toLocaleString('en-IN')}</span>
          </div>
        </div>
      )}

      {/* App Intent Deep-links for Mobile Users */}
      {activeTab === 'intent' && (
        <div className="p-4 sm:p-5 bg-white rounded-2xl border border-stone-200 space-y-3">
          <div className="text-xs font-semibold text-stone-800">
            Tap to open payment directly in your installed UPI app:
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 text-xs">
            <a
              href={upiUri}
              className="p-3 rounded-xl border border-stone-200 hover:border-amber-400 hover:bg-amber-50/50 flex items-center justify-between transition-colors group cursor-pointer"
            >
              <div className="flex items-center gap-2.5">
                <div className="w-8 h-8 rounded-lg bg-blue-50 border border-blue-200 flex items-center justify-center font-bold text-blue-600 text-xs">
                  GPay
                </div>
                <div className="text-left">
                  <div className="font-bold text-stone-900 group-hover:text-amber-900">Google Pay</div>
                  <div className="text-[10px] text-stone-500">Pay directly via GPay</div>
                </div>
              </div>
              <ExternalLink className="w-4 h-4 text-stone-400 group-hover:text-amber-700" />
            </a>

            <a
              href={upiUri}
              className="p-3 rounded-xl border border-stone-200 hover:border-purple-400 hover:bg-purple-50/50 flex items-center justify-between transition-colors group cursor-pointer"
            >
              <div className="flex items-center gap-2.5">
                <div className="w-8 h-8 rounded-lg bg-purple-50 border border-purple-200 flex items-center justify-center font-bold text-purple-600 text-xs">
                  Pe
                </div>
                <div className="text-left">
                  <div className="font-bold text-stone-900 group-hover:text-purple-900">PhonePe</div>
                  <div className="text-[10px] text-stone-500">Pay directly via PhonePe</div>
                </div>
              </div>
              <ExternalLink className="w-4 h-4 text-stone-400 group-hover:text-purple-700" />
            </a>

            <a
              href={upiUri}
              className="p-3 rounded-xl border border-stone-200 hover:border-sky-400 hover:bg-sky-50/50 flex items-center justify-between transition-colors group cursor-pointer"
            >
              <div className="flex items-center gap-2.5">
                <div className="w-8 h-8 rounded-lg bg-sky-50 border border-sky-200 flex items-center justify-center font-bold text-sky-600 text-xs">
                  Paytm
                </div>
                <div className="text-left">
                  <div className="font-bold text-stone-900 group-hover:text-sky-900">Paytm UPI</div>
                  <div className="text-[10px] text-stone-500">Pay directly via Paytm</div>
                </div>
              </div>
              <ExternalLink className="w-4 h-4 text-stone-400 group-hover:text-sky-700" />
            </a>

            <a
              href={upiUri}
              className="p-3 rounded-xl border border-stone-200 hover:border-stone-400 hover:bg-stone-50 flex items-center justify-between transition-colors group cursor-pointer"
            >
              <div className="flex items-center gap-2.5">
                <div className="w-8 h-8 rounded-lg bg-emerald-50 border border-emerald-200 flex items-center justify-center font-bold text-emerald-700 text-xs">
                  BHIM
                </div>
                <div className="text-left">
                  <div className="font-bold text-stone-900 group-hover:text-stone-900">Any UPI App</div>
                  <div className="text-[10px] text-stone-500">Cred, Amazon Pay, Bank apps</div>
                </div>
              </div>
              <ExternalLink className="w-4 h-4 text-stone-400 group-hover:text-stone-700" />
            </a>
          </div>

          <div className="pt-2 text-[11px] text-stone-500 text-center">
            Or manually send ₹{amount.toLocaleString('en-IN')} to UPI ID:{' '}
            <strong className="text-stone-800 font-mono select-all">
              {UPI_PAYEE_DETAILS.upiId}
            </strong>
          </div>
        </div>
      )}

      {/* Step 2: Confirmation / UTR Entry */}
      <div className="p-4 bg-white rounded-xl border border-stone-200 space-y-3">
        <div>
          <label className="block text-xs font-semibold text-stone-800 mb-1">
            After completing payment, enter 12-digit UPI UTR / Reference No. (Optional):
          </label>
          <div className="flex gap-2">
            <input
              type="text"
              placeholder="e.g. 428910294812"
              value={utrNumber}
              maxLength={16}
              onChange={(e) => {
                setUtrNumber(e.target.value.replace(/\s+/g, ''));
                if (utrError) setUtrError(null);
              }}
              className="flex-1 px-3 py-2 bg-stone-50 border border-stone-300 rounded-lg text-xs font-mono focus:bg-white focus:outline-none focus:border-stone-900"
            />
            <button
              type="button"
              onClick={handleConfirmPayment}
              disabled={isProcessing}
              className="px-5 py-2 bg-emerald-600 hover:bg-emerald-700 active:scale-98 text-white rounded-lg text-xs font-bold transition-all shadow-xs flex items-center gap-1.5 cursor-pointer disabled:opacity-50"
            >
              <CheckCircle2 className="w-4 h-4" />
              <span>Verify & Complete</span>
            </button>
          </div>
          {utrError && <p className="text-[11px] text-rose-600 mt-1">{utrError}</p>}
        </div>

        <div className="flex flex-wrap items-center justify-between text-[11px] text-stone-500 gap-2 pt-1 border-t border-stone-100">
          <div className="flex items-center gap-1.5 text-emerald-700">
            <ShieldCheck className="w-3.5 h-3.5" />
            <span>Instant Bank-to-Bank Verification</span>
          </div>
          <div className="text-stone-400">
            No Extra Convenience Fees
          </div>
        </div>
      </div>
    </div>
  );
};
